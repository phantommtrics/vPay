```App.tsx
import React from 'react'
import { MemoryRouter as Router, Routes, Route } from 'react-router-dom'
import { MobileShell } from './components/MobileShell'
import { Home } from './pages/Home'
import { Cards } from './pages/Cards'
import { Fund } from './pages/Fund'
import { Transactions } from './pages/Transactions'
import { Profile } from './pages/Profile'
import { manifest } from './canvas.manifest.js'
// When previewed inside a Magic Patterns "Screen", the iframe URL carries a
// ?mp_screen=<id> param. MemoryRouter ignores the browser URL, so resolve the
// screen's route from the manifest and seed it as the router's initial entry.
function getInitialRoute(): string {
  if (typeof window === 'undefined') return '/'
  const screenId = new URLSearchParams(window.location.search).get('mp_screen')
  if (!screenId) return '/'
  return manifest?.screens?.[screenId]?.route ?? '/'
}
export function App() {
  return (
    <Router initialEntries={[getInitialRoute()]}>
      <MobileShell>
        <Routes>
          <Route path="/" element={<Home />} />
          <Route path="/cards" element={<Cards />} />
          <Route path="/fund" element={<Fund />} />
          <Route path="/transactions" element={<Transactions />} />
          <Route path="/profile" element={<Profile />} />
        </Routes>
      </MobileShell>
    </Router>
  )
}
```

```components/VirtualCard.tsx
import React, { useState } from 'react'
import { motion } from 'framer-motion'
import { Eye, EyeOff, Copy, CheckCircle2 } from 'lucide-react'
import { cn } from '../lib/utils'
interface VirtualCardProps {
  balance: number
  cardNumber: string
  expiry: string
  cvv: string
  isFrozen?: boolean
  className?: string
}
export function VirtualCard({
  balance,
  cardNumber,
  expiry,
  cvv,
  isFrozen = false,
  className,
}: VirtualCardProps) {
  const [showDetails, setShowDetails] = useState(false)
  const [copied, setCopied] = useState(false)
  const handleCopy = (e: React.MouseEvent) => {
    e.stopPropagation()
    navigator.clipboard.writeText(cardNumber.replace(/\s/g, ''))
    setCopied(true)
    setTimeout(() => setCopied(false), 2000)
  }
  return (
    <motion.div
      className={cn(
        'relative w-full aspect-[1.586] rounded-2xl p-6 text-white overflow-hidden shadow-xl transition-all duration-500',
        isFrozen
          ? 'bg-gradient-to-br from-gray-500 via-gray-600 to-gray-700 grayscale-[0.5]'
          : 'bg-gradient-to-br from-emerald-950 via-emerald-800 to-teal-900',
        className,
      )}
      layout
    >
      {/* Background patterns */}
      <div className="absolute inset-0 opacity-20">
        <div className="absolute -top-24 -right-24 w-64 h-64 rounded-full border-[40px] border-white/20 blur-xl" />
        <div className="absolute -bottom-24 -left-24 w-64 h-64 rounded-full border-[40px] border-white/20 blur-xl" />
      </div>

      {/* Glass reflection */}
      <div className="absolute inset-0 bg-gradient-to-tr from-white/10 via-transparent to-transparent opacity-50" />

      {isFrozen && (
        <div className="absolute inset-0 bg-black/20 backdrop-blur-[2px] z-10 flex items-center justify-center">
          <div className="bg-white/20 px-4 py-2 rounded-full backdrop-blur-md border border-white/30">
            <span className="font-medium text-sm tracking-widest uppercase">
              Card Frozen
            </span>
          </div>
        </div>
      )}

      <div className="relative z-20 h-full flex flex-col justify-between">
        <div className="flex justify-between items-start">
          <div>
            <p className="text-emerald-200/80 text-xs font-medium uppercase tracking-wider mb-1">
              Available Balance
            </p>
            <h3 className="text-2xl font-bold tracking-tight">
              ${balance.toFixed(2)}
            </h3>
          </div>

          {/* Mock Visa/Mastercard Logo */}
          <div className="flex items-center">
            <div className="w-8 h-8 bg-red-500/80 rounded-full mix-blend-multiply" />
            <div className="w-8 h-8 bg-yellow-500/80 rounded-full -ml-4 mix-blend-multiply" />
          </div>
        </div>

        <div className="space-y-4">
          <div className="flex items-center justify-between">
            <div className="font-mono text-lg tracking-[0.15em] drop-shadow-sm">
              {showDetails
                ? cardNumber
                : `•••• •••• •••• ${cardNumber.slice(-4)}`}
            </div>
            <button
              onClick={(e) => {
                e.stopPropagation()
                setShowDetails(!showDetails)
              }}
              className="p-2 hover:bg-white/10 rounded-full transition-colors"
            >
              {showDetails ? <EyeOff size={18} /> : <Eye size={18} />}
            </button>
          </div>

          <div className="flex justify-between items-end">
            <div className="flex space-x-6">
              <div>
                <p className="text-[10px] text-emerald-200/80 uppercase tracking-wider mb-0.5">
                  Valid Thru
                </p>
                <p className="font-mono text-sm">{expiry}</p>
              </div>
              <div>
                <p className="text-[10px] text-emerald-200/80 uppercase tracking-wider mb-0.5">
                  CVV
                </p>
                <p className="font-mono text-sm">{showDetails ? cvv : '•••'}</p>
              </div>
            </div>

            <button
              onClick={handleCopy}
              className="flex items-center space-x-1.5 bg-white/10 hover:bg-white/20 px-3 py-1.5 rounded-full transition-colors text-xs font-medium backdrop-blur-sm"
            >
              {copied ? (
                <CheckCircle2 size={14} className="text-green-400" />
              ) : (
                <Copy size={14} />
              )}
              <span>{copied ? 'Copied' : 'Copy'}</span>
            </button>
          </div>
        </div>
      </div>
    </motion.div>
  )
}
```

```useScreenInit.js
import { useMemo } from 'react'
import { manifest } from './canvas.manifest.js'

export function useScreenInit() {
  return useMemo(() => {
    if (typeof window === 'undefined') return {}
    const screenId = new URLSearchParams(window.location.search).get('mp_screen')
    if (!screenId) return {}
    return manifest?.screens?.[screenId]?.state ?? {}
  }, [])
}
```
